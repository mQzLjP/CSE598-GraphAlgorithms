\section{Improvements}

We want to improve $L$ so that $L = O(n\log (nC_{\max}))$.

The current plan is that we do multiple rounds of Algorithm~\ref{alg:min-cost-flow}, with decreasing $\varepsilon$ values, e.g. $\varepsilon = C_{\max}, C_{\max}/2, C_{\max}/4, \ldots, 1/(n+1)$.
But how can we continue from a solution with $\varepsilon$ to a solution with $\varepsilon/2$? We can use the following lemma.

\begin{lemma}
	Fix $(f,h)$.
	Then $x$ is a flow in $G_{f,h}$ routing $\ex_f = b+f^{\text{net}}$ if and only if $f+x$ is a flow in $G$ routing $b$.

	Moreover, $\cost_G(f+x) = \cost_{G}(f) + \cost_{G_{f,h}}(x) + \sum_{v \in V} h(v) \cdot \ex_f(v)$.
\end{lemma}
\begin{proof}
	Since $(f+x)^{\text{net}} = f^{\text{net}} + x^{\text{net}}$, we have
	\[ b+(f+x)^{\text{net}} = b+f^{\text{net}} + x^{\text{net}} = \ex_f + x^{\text{net}}. \]
	So $f+x$ is a flow in $G$ routing $b$ if and only if $x$ is a flow in $G_{f,h}$ routing $\ex_f$.

	Also, as $ \cost_G(f+x) = \cost_G(f)+\cost_{G_f}(x) $ by construction, and by definition, $\forall e = (u,v) \in E: c_{f,h}(e) = c_f(e) - h(u) + h(v)$, we have

	\begin{align*}
		\sum_{e \in E} c_f(e) \cdot x(e) & = \sum_{e = (u,v) \in E} (c_{f,h}(e) + h(u) - h(v)) \cdot x(e)                                    \\
		                                 & = \cost_{G_{f,h}}(x) + \sum_{e=(u,v) \in E} (h(u) - h(v)) \cdot x(e)                              \\
		                                 & = \cost_{G_{f,h}}(x) + \sum_{v \in V} h(v) \cdot (x^{\text{out}}(v) - x^{\text{in}}(v))           \\
		                                 & = \cost_{G_{f,h}}(x) - \sum_{v\in V} h(v) \cdot x^{\text{net}}(v)                                 \\
		                                 & = \cost_{G_{f,h}}(x) + \sum_{v\in V} h(v) \cdot \ex_f(v) \quad \text{(since $x$ routes $\ex_f$).}
	\end{align*}

	Therefore, we have
	\[ \cost_G(f+x) = \cost_G(f) + \cost_{G_{f,h}}(x) + \sum_{v\in V} h(v) \cdot \ex_f(v). \]
\end{proof}

Then we want to find suitable $(f,h)$.

\begin{definition}
	For $G$,
	\[ \Delta(G) = \max(\{0\}\cup\{-c(e)\}_{e \in E}) \]

	Then $(f,h)$ is $\varepsilon$-optimal if $\forall e, c_{f,h}(e) \geq -\varepsilon \iff \Delta(G_{f,h}) \leq \varepsilon$.
    
\end{definition}

\begin{theorem} \label{thm:refine}
	Given $G$ with (not necessarily integral) costs, there exists an algorithm $\mathsf{Refine}(G)$ that returns $(f,h)$ such that
	\[ \Delta(G_{f,h}) \leq \frac{\Delta(G)}{2}\]
	in time $O(mn^2)$.
\end{theorem}

For now we assume Theorem~\ref{thm:refine} is true, and we will prove it later.

\subsection{Cost Scaling}

Consider the following Successive Approximation Algorithm:

\begin{algorithm}[H]
	\caption{Successive Approximation}
	\label{alg:successive-approximation}
	Find flow $f_0$ routing $b$ (or report infeasible) in $G$\;
	Set $h_0\gets 0$, $G^{(0)} \gets G$\;
	\ForEach{$i = 1,2,\ldots$ until $\Delta(G^{(i)}) < 1/n$}{
	  $(f_{i},h_{i}) \gets \mathsf{Refine}(G^{(i)})$\;
		$G^{(i)} \gets G^{(i-1)}_{f_{i},h_{i}}$\;
	}
	\Return $f^* = f_0 + f_1 + \cdots + f_i$ and $h^* = h_0+h_1+\cdots+h_i$\;
\end{algorithm}

For correctness, it suffices to prove the following lemma:
\begin{lemma}
	For $(f,h)$ in graph $H$, let $H' = H_{f,h}$.
	Let $(f',h')$ in $H'$.

	Then $H'_{f',h'} = H_{f+f',h+h'}$.
\end{lemma}

The proof is simply calculation regarding the definitions of $H_{f,h}$ and $H'_{f',h'}$, so we omit it.
    
Then $G^{(i)} = G_{f^*,h^*}$. We see that $f^*$ is the minimum cost flow routing $b$ because $f^*$ routes $b$, and $\Delta(G_{f^*,h^*}) < 1/n$ implies that $f^*$ is optimal by Lemma~\ref{lem:eps-optimal}.

If Theorem~\ref{thm:refine} is true, the running time of Algorithm~\ref{alg:successive-approximation} is $O(mn^2 \log(nC_{\max}))$ as desired.

\subsection{Refine Algorithm}

The algorithm $\mathsf{Refine}(G)$ is as follows:

\begin{algorithm}[H]
	\caption{$\mathsf{Refine}(G)$}
	\label{alg:refine}
	$\varepsilon \gets \Delta(G)/2$\;
	$b\gets 0$, $h\gets 0$\;
	$f(e)\gets\begin{cases}
		U(e) & \text{if $c(e) < 0$} \\
		0    & \text{otherwise}
	\end{cases}$\;
	\While{$\exists$ active $u$}{
		\eIf{$\exists$ admissible edge $(u,v)$}{
			Push $\min\{U_f(u,v), \ex_f(u)\}$ units of flow from $u$ to $v$\;
		}{
			$h(u) \gets h(u) + \varepsilon$\;
		}
	}
	\Return $(f,h)$\;
\end{algorithm}

Other than value of $\varepsilon$, the algorithm is the same as Algorithm~\ref{alg:min-cost-flow}.
The invariants remain the same, so we omit the correctness proof here.

\begin{lemma}
    The maximum height of any vertex is at most $1.5 n \cdot \Delta(G)$.
\end{lemma}
\begin{proof}
	Consider $u$ with excess $\ex_f(u) > 0$.
	Then $u$ is active, and there exists a path $P$ from $u$ to some deficit vertex $z$ in $G_f$ as we proved before.
	So,
	\[ h(u) \le h(z) + c_f(P) + |P| \cdot \varepsilon \le 0 + n \cdot \Delta(G) + n \cdot \frac{\Delta(G)}{2} = 1.5 n \cdot \Delta(G). \]
\end{proof}

Therefore, the number of relabel operations for a single vertex is $L = O(n \cdot \Delta(G)/\varepsilon) = O(n)$. Similar to what we did for Max Flow, we can show that the total number of relabels are $O(nL) = O(n^2)$, the number of unsaturating pushes is at most $O(mL) = O(nm)$, and the number of saturating pushes is at most $O(mnL)=O(n^2 m)$. 
So the total running time of $\mathsf{Refine}(G)$ is $O(n^2 m)$, which proves Theorem~\ref{thm:refine}.
